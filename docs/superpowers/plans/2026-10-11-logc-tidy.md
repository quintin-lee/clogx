```python
import sys

SRC = "core/log.c"
raw = open(SRC).read().split("\n")
if raw and raw[-1] == "":
    raw.pop()  # 去掉末尾换行符产生的空元素；行号 1..1765
assert len(raw) == 1765, "expected 1765 lines, got %d" % len(raw)

# (name, start, end, must_contain) —— 行号 1-indexed 闭区间，首尾空行自动剥离
SPECS = [
    ("head",      1,    82,   ["g_span_depth = 0;"]),
    ("spanhelp",  83,   157,  ["span_new_span_id"]),
    ("fwdhex",    159,  159,  ["parse_hex_nibble"]),
    ("strerror",  160,  193,  ["log_strerror(int err)"]),
    ("fallback",  194,  203,  ["log_get_async_fallback_cb"]),
    ("module",    204,  234,  ["log_get_module"]),
    ("sink1",     235,  261,  ["log_remove_sink"]),
    ("initint",   262,  308,  ["logger_init_internal"]),
    ("write",     309,  628,  ["logger_write_kv_internal", "Internal write path"]),
    ("stats",     629,  638,  ["log_get_stats"]),
    ("mdcstate",  639,  648,  ["g_thread_context"]),
    ("mdc",       649,  719,  ["log_clear_thread_context"]),
    ("trace",     720,  811,  ["clog_span_end"]),
    ("redact",    812, 1123,  ["redact_apply_all_heap_locked"]),
    ("spanexp",   1124, 1275, ["parse_hex_nibble", "clog_span_join"]),
    ("atfork",    1276, 1350, ["register_atfork", "#endif"]),
    ("life",      1351, 1476, ["log_reload"]),
    ("multi",     1477, 1655, ["logger_create_from_config", "Multi-instance API"]),
    ("sink2",     1655, 1673, ["logger_remove_sink"]),
    ("multi2",    1674, 1740, ["logger_config_get"]),
    ("mutex",     1741, 1765, ["clog_mutex_unlock"]),
]

def get(a, b):
    blk = raw[a - 1:b]
    while blk and not blk[0].strip():
        blk.pop(0)
    while blk and not blk[-1].strip():
        blk.pop()
    assert blk, "empty block %d-%d" % (a, b)
    return blk

blocks = {}
covered = set()
for name, a, b, must in SPECS:
    blk = get(a, b)
    text = "\n".join(blk)
    for m in must:
        assert m in text, "block %s missing %r" % (name, m)
    assert name not in blocks
    blocks[name] = blk
    covered.update(range(a, b + 1))

# 158 是 span helper 与 fwd 声明之间的空行，唯一允许的未覆盖行
assert covered == set(range(1, 1766)) - {158}, "coverage gap"

def banner(title):
    bar = "/* " + "═" * 78
    return [bar, " *  " + title, " " + "═" * 78 + " */", ""]

# (block, banner-or-None)：banner 插在该 block 之前
ORDER = [
    ("head", None),
    ("strerror", None),
    ("write", None),
    ("initint", "Singleton lifecycle (init/destroy/flush/reload, stats, fallback)"),
    ("life", None),
    ("fallback", None),
    ("stats", None),
    ("multi", None),
    ("multi2", None),
    ("sink1", "Sink management (singleton + multi-instance)"),
    ("sink2", None),
    ("spanhelp", "Trace / Span (helpers + context + export)"),
    ("fwdhex", None),
    ("trace", None),
    ("spanexp", None),
    ("redact", "Record-time redaction (rule table + appliers)"),
    ("module", "MDC thread context + module name"),
    ("mdcstate", None),
    ("mdc", None),
    ("atfork", "Platform adaptation (fork + mutex wrappers)"),
    ("mutex", None),
]

out = []
for name, b in ORDER:
    if b:
        out.extend(banner(b))
    out.extend(blocks[name])
    out.append("")

open(SRC, "w").write("\n".join(out).rstrip("\n") + "\n")
print("reorder OK: %d blocks, %d lines" % (len(blocks), len(out)))
```
